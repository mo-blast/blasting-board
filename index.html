// MEMASTIKAN SELURUH ELEMEN HTML SELESAI DIMUAT BROWSER SEBELUM JAVASCRIPT BERJALAN window.onload = function() { // 1. DATA DEFAULT & CACHE LOKAL BROWSER const defaultData = { date: "19 / 08 / 2026", time: "14.45 WITA", location: "Harapan West Quary" }; const localCached = localStorage.getItem('vale_blasting_local'); const initialData = localCached ? JSON.parse(localCached) : defaultData; // Fungsi menggambar data ke teks layar function renderToUI(data) { if(!data) return; const lblDate = document.getElementById('lbl-date'); const lblTime = document.getElementById('lbl-time'); const lblLocation = document.getElementById('lbl-location'); if(lblDate) lblDate.innerText = data.date || ""; if(lblTime) lblTime.innerText = data.time || ""; if(lblLocation) lblLocation.innerText = data.location || ""; const inpDate = document.getElementById('inp-date'); const inpTime = document.getElementById('inp-time'); const inpLocation = document.getElementById('inp-location'); if(inpDate) inpDate.value = data.date || ""; if(inpTime) inpTime.value = data.time || ""; if(inpLocation) inpLocation.value = data.location || ""; } // 2. LOGIKA PENYARINGAN HALAMAN (MONITOR VS ADMIN HP) function aturTampilanHalaman() { const hpPanel = document.getElementById('hp-panel'); const mainDash = document.getElementById('main-dash'); const audioControl = document.getElementById('audio-control'); if (window.location.search.includes('admin')) { if(hpPanel) hpPanel.style.display = 'block'; if(mainDash) mainDash.style.display = 'none'; if(audioControl) audioControl.style.display = 'none'; } else { if(mainDash) mainDash.style.display = 'block'; if(hpPanel) hpPanel.style.display = 'none'; if(audioControl) audioControl.style.display = 'block'; } } // Langsung eksekusi visual awal (Instan 0 detik) renderToUI(initialData); aturTampilanHalaman(); // 3. KONEKSI KE SERVER INTERNET GUN DATABASE const gun = Gun({ peers: [ 'https://herokuapp.com', 'https://herokuapp.com', 'https://herokuapp.com' ], localStorage: true }); const blastingNode = gun.get('vale_blasting_mo_blast_2026'); // Mendengarkan pembaruan data secara live dari internet blastingNode.on(function(data) { if(data && (data.date || data.time || data.location)) { renderToUI(data); localStorage.setItem('vale_blasting_local', JSON.stringify(data)); } }); // Menghubungkan fungsi klik tombol ke form HTML secara aman const formAdmin = document.querySelector('form'); if(formAdmin) { formAdmin.onsubmit = function(e) { e.preventDefault(); const d = document.getElementById('inp-date').value; const t = document.getElementById('inp-time').value; const l = document.getElementById('inp-location').value; const newData = { date: d, time: t, location: l }; localStorage.setItem('vale_blasting_local', JSON.stringify(newData)); renderToUI(newData); blastingNode.put(newData, function(ack){ alert("Data Berhasil Sinkron ke Monitor Utama!"); }); }; } }; // KONTROL AUDIO SIRINE (Diletakkan di luar agar tombol HTML bisa mendeteksi langsung) const audio = document.getElementById("sirineAudio"); const btn = document.getElementById("audio-control"); function togglePlay() { if (audio.paused) { audio.play().catch(e => console.log("Izin audio aktif")); btn.innerText = "⏸ Matikan Sirine"; } else { audio.pause(); btn.innerText = "🔊 Aktifkan Sirine"; } }
